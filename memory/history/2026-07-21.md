### 3. 생성 스크립트 (Blender Python Script)

```python
import bpy
import bmesh

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes["Principled BSDF"].inputs[0].default_value = color # Base Color
    nodes["Principled BSDF"].inputs[7].default_value = 0.1 # Roughness
    return mat

def create_dummy():
    # 기존 오브젝트 삭제
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

    # 재질 생성
    yellow_mat = create_material("Dummy_Yellow", (1, 0.7, 0, 1))
    black_mat = create_material("Dummy_Black", (0.05, 0.05, 0.05, 1))

    # 1. 머리 (Head)
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.25, location=(0, 0, 1.7))
    head = bpy.context.active_object
    head.scale[2] = 1.2
    head.data.materials.append(yellow_mat)

    # 2. 목 (Neck)
    bpy.ops.mesh.primitive_cylinder_add(radius=0.05, depth=0.15, location=(0, 0, 1.45))
    neck = bpy.context.active_object
    neck.data.materials.append(black_mat)

    # 3. 가슴 (Torso)
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 1.1))
    torso = bpy.context.active_object
    torso.scale = (0.25, 0.15, 0.3)
    torso.data.materials.append(yellow_mat)

    # 4. 골반 (Pelvis)
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0.75))
    pelvis = bpy.context.active_object
    pelvis.scale = (0.2, 0.15, 0.15)
    pelvis.data.materials.append(yellow_mat)

    # 5. 팔 (Arms - 간단하게 한쪽만 생성 후 대칭 가능)
    def create_arm(side):
        x_pos = 0.4 * side
        # 어깨 관절
        bpy.ops.mesh.primitive_uv_sphere_add(radius=0.06, location=(x_pos*0.7, 0, 1.3))
        bpy.context.active_object.data.materials.append(black_mat)
        # 상완
        bpy.ops.mesh.primitive_cylinder_add(radius=0.06, depth=0.35, location=(x_pos, 0, 1.15))
        arm_up = bpy.context.active_object
        arm_up.rotation_euler[1] = 0.2 * side
        arm_up.data.materials.append(yellow_mat)
        # 전완
        bpy.ops.mesh.primitive_cylinder_add(radius=0.05, depth=0.35, location=(x_pos*1.3, 0, 0.85))
        arm_low = bpy.context.active_object
        arm_low.rotation_euler[1] = 0.4 * side
        arm_low.data.materials.append(yellow_mat)

    create_arm(1)  # 오른쪽 팔
    create_arm(-1) # 왼쪽 팔

    # 6. 다리 (Legs)
    def create_leg(side):
        x_pos = 0.12 * side
        # 허벅지
        bpy.ops.mesh.primitive_cylinder_add(radius=0.08, depth=0.45, location=(x_pos, 0, 0.45))
        leg_up = bpy.context.active_object
        leg_up.data.materials.append(yellow_mat)
        # 무릎 관절
        bpy.ops.mesh.primitive_uv_sphere_add(radius=0.07, location=(x_pos, 0, 0.2))
        bpy.context.active_object.data.materials.append(black_mat)
        # 종아리
        bpy.ops.mesh.primitive_cylinder_add(radius=0.07, depth=0.4, location=(x_pos, 0, -0.05))
        leg_low = bpy.context.active_object
        leg_low.data.materials.append(yellow_mat)
        # 발
        bpy.ops.mesh.primitive_cube_add(size=0.1, location=(x_pos, 0.05, -0.3))
        foot = bpy.context.active_object
        foot.scale = (0.8, 1.5, 0.5)
        foot.data.materials.append(yellow_mat)

    create_leg(1)
    create_leg(-1)

    # 모든 물체 부드럽게
    for obj in bpy.data.objects:
        if obj.type == 'MESH':
            bpy.context.view_layer.objects.active = obj
            bpy.ops.object.shade_smooth()

create_dummy()
```
